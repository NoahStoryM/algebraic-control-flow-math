;; 试验：Noah 22:17 的程序，基于 ambKanren 的 demo.rkt（一字不动，只把 typed/amb 换成第五篇的三个定义）
(require "demo-minimal.rkt" racket/list)

;; demo.rkt 里没有 project；这是 ambKanren main.rkt 里的定义，照抄
(define-syntax project
  (syntax-rules ()
    [(_ () g ...) (all g ...)]
    [(_ (x ...) g ...)
     (λ (s) (let ([x (walk* x s)] ...) ((all g ...) s)))]))

(define (pyth? a b c) (= (+ (* a a) (* b b)) (* c c)))

;; pytho：要用到三个数的值，官方的办法是 project
(define (pytho a b c)
  (project (a b c)
    (if (pyth? a b c) succeed fail)))

;; betweeno：lo、hi 已知，x 依次取其间的数（conde + 递归）
(define (betweeno lo x hi)
  (project (lo hi)
    (if (> lo hi)
        fail
        (conde [(== x lo)]
               [else (betweeno (add1 lo) x hi)]))))

;; (≤ l a b c h)：一段一段接起来
(define (≤ lo . rest)
  (define hi (last rest))
  (let loop ([prev lo] [xs (drop-right rest 1)])
    (if (null? xs)
        succeed
        (all (betweeno prev (car xs) hi) (loop (car xs) (cdr xs))))))

;; 2. 官方第一版的写法：一个查询变量，fresh 引入三条边
(define (right-triangles l h)
  (run* (t)
    (fresh (a b c)
      (== t (list a b c))
      (≤ l a b c h)
      (pytho a b c))))
(printf "run* (t) + fresh：~s\n" (right-triangles 1 20))
